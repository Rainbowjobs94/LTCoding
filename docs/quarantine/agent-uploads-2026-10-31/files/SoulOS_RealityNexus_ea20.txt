    '[SEC] SELinux enforcing, sVirt labels applied.',

        entryValue: newValue,
        currentValue: newValue,
        ceiling: amount * 10,

        active: true,
        entryValue: newValue,
        currentValue: newValue,

                          {item.status}
                        </span>
                      </div>
                      <p className="text-xs text-slate-500 dark:text-slate-400 mt-2 leading-relaxed">{item.desc}</p>
                    </div>
                  ))}
                </div>
              </div>
            </div>
          )}

        </main>

        
        {/* Share Side Panel */}
        {sharePanelOpen && (
          <div className="fixed inset-0 z-50 flex justify-end">
            <div className="absolute inset-0 bg-black/60 backdrop-blur-sm" onClick={() => setSharePanelOpen(false)} />
            <div className="w-full max-w-sm h-full shadow-2xl relative flex flex-col animate-in slide-in-from-right duration-300 bg-white dark:bg-slate-900 border-l border-slate-200 dark:border-slate-800">
              <div className="p-4 border-b border-slate-200 dark:border-slate-800 flex items-center justify-between">
                <h3 className="font-bold text-lg text-slate-900 dark:text-white">Share this post</h3>
                <button onClick={() => setSharePanelOpen(false)} className="p-2 hover:bg-slate-100 dark:hover:bg-slate-800 rounded-full text-slate-500 dark:text-slate-400 transition-colors">
                  <X className="w-5 h-5" />
                </button>
              </div>
              
              <div className="p-6 grid grid-cols-2 gap-4">
                {[
                  { label: 'Copy Link', action: () => { showToast('Link Copied!'); setSharePanelOpen(false); } },
                  { label: 'Send in DM', action: () => { showToast('DM Modal opened'); setSharePanelOpen(false); } },
                  { label: 'Share to Feed', action: () => { showToast('Reposted to your feed'); setSharePanelOpen(false); } },
                  { label: 'Share to Public', action: () => { showToast('Shared to public tier'); setSharePanelOpen(false); } },
                ].map((opt, i) => (
                  <button key={i} onClick={opt.action} className="p-4 rounded-xl border border-slate-200 hover:bg-slate-50 dark:border-slate-700 dark:hover:bg-slate-800 flex flex-col items-center justify-center gap-2 text-sm text-slate-700 dark:text-slate-300 transition-colors">
                    <Share className="w-5 h-5 text-purple-500 dark:text-purple-400" />
                    {opt.label}
                  </button>
                ))}
              </div>

              <div className="mt-auto p-6 bg-slate-50 dark:bg-slate-950 border-t border-slate-200 dark:border-slate-800">
                <p className="text-xs text-slate-500 font-mono mb-2">POST PREVIEW:</p>
                <p className="text-sm text-slate-700 dark:text-slate-300 line-clamp-2 italic">
                  "{posts.find(p => p.id === selectedPostId)?.content || '...'}"
                </p>
              </div>
            </div>
          </div>
        )}

        {/* Action Modals (Mint, Shift Mine, Buy SC) */}
        {activeModal && (
          <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
            <div className="absolute inset-0 bg-black/80 backdrop-blur-sm" onClick={() => setActiveModal(null)} />
            
            <div className="relative w-full max-w-md rounded-3xl border shadow-2xl overflow-hidden animate-in zoom-in-95 duration-200 bg-white border-slate-200 dark:bg-slate-900 dark:border-slate-700">
              
              {activeModal === 'mint' && (
                <div className="p-6">
                  <h3 className="text-xl font-bold mb-4 flex items-center gap-2 text-slate-900 dark:text-white"><Coins className="text-yellow-500" /> Mint This Post as NFT</h3>
                  <div className="p-4 bg-slate-50 dark:bg-slate-950 rounded-xl mb-6 text-sm text-slate-600 dark:text-slate-400 border border-slate-200 dark:border-slate-800">
                    <p className="italic mb-2 line-clamp-3">"{posts.find(p => p.id === selectedPostId)?.content}"</p>
                    <p className="text-yellow-600 dark:text-yellow-500 font-bold mt-4">Minting fee: 15 RC</p>
                  </div>
                  <p className="text-xs text-slate-500 mb-6 leading-relaxed">
                    This post will become a permanent on-chain NFT and will be listed in the LTNFThifty marketplace. Once minted, this post cannot be deleted.
                  </p>
                  <div className="flex gap-3">
                    <button onClick={() => setActiveModal(null)} className="flex-1 py-2.5 rounded-xl border border-slate-300 text-slate-700 dark:border-slate-700 dark:text-slate-300 font-medium hover:bg-slate-50 dark:hover:bg-slate-800 transition-colors">Cancel</button>
                    <button onClick={confirmMint} className="flex-1 py-2.5 rounded-xl bg-yellow-500 hover:bg-yellow-600 text-black font-bold transition-colors">Mint Now</button>
                  </div>
                </div>
              )}

              {activeModal === 'shiftMine' && (
                <div className="p-6 border-t-4 border-red-500">
                  <h3 className="text-xl font-bold mb-2 flex items-center gap-2 text-red-600 dark:text-red-500"><AlertTriangle /> PERMANENT ACTION</h3>
                  <p className="text-sm text-slate-700 dark:text-slate-300 font-medium mb-4">Read before continuing.</p>
                  
                  <ul className="text-xs text-slate-600 dark:text-slate-400 space-y-2 mb-6 list-disc pl-4">
                    <li>Shift Mining registers this post as permanent proof-of-contribution.</li>
                    <li>This post will NEVER be deleted and cannot be edited.</li>
                    <li>Its content hash is recorded on-chain.</li>
                    <li className="text-purple-600 dark:text-purple-400 font-bold">Reward: SC based on content score &amp; RSNX points.</li>
                  </ul>

                  <div className="mb-6">
                    <label className="text-xs font-mono text-slate-500 mb-2 block">Type CONFIRM to proceed:</label>
                    <input type="text" placeholder="CONFIRM" className="w-full bg-slate-50 border border-slate-200 text-slate-900 dark:bg-slate-950 dark:border-slate-800 dark:text-white rounded-lg px-4 py-2 text-sm focus:border-red-500 outline-none" id="confirmInput" onChange={(e) => {
                      const btn = document.getElementById('mineBtn');
                      if (btn) btn.disabled = e.target.value !== 'CONFIRM';
                    }}/>
                  </div>

                  <div className="flex gap-3">
                    <button onClick={() => setActiveModal(null)} className="flex-1 py-2.5 rounded-xl border border-slate-300 text-slate-700 dark:border-slate-700 dark:text-slate-300 font-medium hover:bg-slate-50 dark:hover:bg-slate-800 transition-colors">Cancel</button>
                    <button id="mineBtn" disabled onClick={confirmShiftMine} className="flex-1 py-2.5 rounded-xl bg-red-600 disabled:bg-slate-300 disabled:text-slate-500 dark:disabled:bg-slate-800 dark:disabled:text-slate-500 text-white font-bold transition-colors">Mine Post</button>
                  </div>
                </div>
              )}

              {activeModal === 'buyFiat' && (
                <div className="p-6">
                  <h3 className="text-xl font-bold mb-4 text-slate-900 dark:text-white">Buy SC via Fiat</h3>
                  <div className="space-y-4 mb-6">
                    <div>
                      <label className="text-xs text-slate-500 dark:text-slate-400">Amount (USD)</label>
                      <input type="number" defaultValue="25" id="fiatAmount" className="w-full bg-slate-50 border border-slate-200 text-slate-900 dark:bg-slate-950 dark:border-slate-800 dark:text-white rounded-lg px-4 py-2 mt-1 outline-none focus:border-blue-500" />
                    </div>
                    <div className="p-3 rounded-lg bg-blue-50 border border-blue-200 text-blue-800 dark:bg-blue-500/10 dark:border-blue-500/20 text-xs dark:text-blue-300">
                      <strong>Rule 1:</strong> You will receive 2x in-network value instantly.
                    </div>
                  </div>
                  <div className="flex gap-3">
                    <button onClick={() => setActiveModal(null)} className="flex-1 py-3 border border-slate-300 text-slate-700 dark:border-slate-700 dark:text-slate-300 rounded-xl font-medium">Cancel</button>
                    <button onClick={() => handleBuySC('fiat', Number(document.getElementById('fiatAmount').value))} className="flex-1 py-3 bg-blue-600 hover:bg-blue-700 text-white rounded-xl font-bold transition-colors">Confirm Purchase</button>
                  </div>
                </div>
              )}

              {activeModal === 'buyEth' && (
                <div className="p-6">
                  <h3 className="text-xl font-bold mb-4 text-emerald-600 dark:text-emerald-400">Buy SC via Ethereum</h3>
                  <div className="space-y-4 mb-6">
                    <div>
                      <label className="text-xs text-slate-500 dark:text-slate-400">Amount (ETH)</label>
                      <input type="number" step="0.1" defaultValue="0.1" id="ethAmount" className="w-full bg-slate-50 border border-slate-200 text-slate-900 dark:bg-slate-950 dark:border-slate-800 dark:text-white rounded-lg px-4 py-2 mt-1 outline-none focus:border-emerald-500" />
                    </div>
                    <div className="p-3 rounded-lg bg-emerald-50 border border-emerald-200 text-emerald-800 dark:bg-emerald-500/10 dark:border-emerald-500/20 text-xs dark:text-emerald-300">
                      <strong>Rule 3:</strong> ETH entry grants instant On-Chain status. No treasury cut.
                    </div>
                  </div>
                  <div className="flex gap-3">
                    <button onClick={() => setActiveModal(null)} className="flex-1 py-3 border border-slate-300 text-slate-700 dark:border-slate-700 dark:text-slate-300 rounded-xl font-medium">Cancel</button>
                    <button onClick={() => handleBuySC('eth', Number(document.getElementById('ethAmount').value))} className="flex-1 py-3 bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl font-bold transition-colors">Confirm ETH Transaction</button>
                  </div>
                </div>
              )}

            </div>
          </div>
        )}
      </div>
    </div>
  );
}
